# Live Demo Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Visitors to the demo site can add their own receipt photo and get the real model's answer, read on the author's PC while the author runs `scripts/live-demo.ps1`; the PC keeps nothing.

**Architecture:** The API project gets a second, minimal startup path (`LiveDemo/LiveDemoApp.cs`, chosen by `--LiveDemo:Enabled=true`) that registers only the Ollama extractor and two endpoints (`GET /api/live`, `POST /api/extract`), with CORS for the demo site, a per-visitor rate limit, a one-at-a-time GPU gate with a budget, and a self-stop timer. No database or storage is registered, so nothing can be written. On the web side, `web/src/demo/liveReader.ts` talks to it (status check, one-at-a-time queue, error messages) and `demoApi` routes visitors' own photos through it. Tailscale Funnel exposes port 5090; `VITE_LIVE_URL` tells the Pages build where.

**Tech Stack:** ASP.NET Core (.NET 10) minimal APIs, `System.Threading.RateLimiting`, xUnit + `Microsoft.AspNetCore.TestHost`; React 19 + TypeScript + Vite 8, vitest; PowerShell; Tailscale Funnel.

**Spec:** `docs/superpowers/specs/2026-09-28-live-demo-design.md` (read it first).

```csharp
namespace ReceiptExtractor.Api.Endpoints;

/// <summary>What counts as a receipt photo: for uploads, replacing an image, and the live demo.</summary>
public static class UploadRules
{
    public const long MaxBytes = 15 * 1024 * 1024;

    public static readonly IReadOnlyDictionary<string, (string Ext, string ContentType)> AllowedFiles =
        new Dictionary<string, (string Ext, string ContentType)>(StringComparer.OrdinalIgnoreCase)
        {
            [".jpg"] = (".jpg", "image/jpeg"),
            [".jpeg"] = (".jpg", "image/jpeg"),
            [".png"] = (".png", "image/png"),
        };

    /// <summary>Checks the file's first bytes: JPEG starts with FF D8 FF, PNG with 89 50 4E 47.</summary>
    public static async Task<bool> LooksLikeImageAsync(IFormFile file, CancellationToken ct)
    {
        var header = new byte[4];
        await using var stream = file.OpenReadStream();
        var read = await stream.ReadAtLeastAsync(header, header.Length, throwOnEndOfStream: false, ct);
        if (read < 3) return false;
        var isJpeg = header[0] == 0xFF && header[1] == 0xD8 && header[2] == 0xFF;
        var isPng = read == 4 && header[0] == 0x89 && header[1] == 0x50 && header[2] == 0x4E && header[3] == 0x47;
        return isJpeg || isPng;
    }
}
```

```csharp
using Microsoft.Extensions.Options;

namespace ReceiptExtractor.Api.LiveDemo;

/// <summary>
/// The GPU reads one photo at a time. A few more may wait their turn; past that, visitors are told it's busy.
/// At most <see cref="LiveDemoOptions.Budget"/> reads per run (counted in memory, so a new run starts a new budget).
/// </summary>
public sealed class ReadGate(IOptions<LiveDemoOptions> options)
{
    public enum Refusal { Busy, BudgetUsed }

    /// <summary>Either a turn (dispose it when the read is done) or why there isn't one.</summary>
    public sealed record Result(IDisposable? Turn, Refusal? Refused);

    private readonly SemaphoreSlim _gpu = new(1, 1);
    private int _inLine; // being read + waiting
    private int _used;

    public int InLine => Volatile.Read(ref _inLine);
    public int Used => Volatile.Read(ref _used);
    public int Left => Math.Max(0, options.Value.Budget - Used);

    public async Task<Result> WaitTurnAsync(CancellationToken ct)
    {
        var o = options.Value;
        if (Used >= o.Budget)
            return new(null, Refusal.BudgetUsed);

        if (Interlocked.Increment(ref _inLine) > o.MaxWaiting + 1)
        {
            Interlocked.Decrement(ref _inLine);
            return new(null, Refusal.Busy);
        }

        try
        {
            await _gpu.WaitAsync(ct);
        }
        catch (OperationCanceledException)
        {
            Interlocked.Decrement(ref _inLine); // the visitor left while waiting
            throw;
        }

        if (Interlocked.Increment(ref _used) > o.Budget) // the budget ran out while this one waited
        {
            Interlocked.Decrement(ref _used);
            Release();
            return new(null, Refusal.BudgetUsed);
        }

        return new(new Turn(this), null);
    }

    private void Release()
    {
        Interlocked.Decrement(ref _inLine);
        _gpu.Release();
    }

    private sealed class Turn(ReadGate gate) : IDisposable
    {
        private int _done;

        public void Dispose()
        {
            if (Interlocked.Exchange(ref _done, 1) == 0)
                gate.Release();
        }
    }
}
```

```csharp
using System.Diagnostics;
using System.Security.Cryptography;
using Microsoft.Extensions.Options;
using ReceiptExtractor.Api.Endpoints;
using ReceiptExtractor.Api.Extraction;
using ReceiptExtractor.Api.Validation;

namespace ReceiptExtractor.Api.LiveDemo;

/// <summary>GET /api/live: the demo site's "is the author's PC on?" check.</summary>
public sealed record LiveStatus(string Model, int? PerVisitorPerHour);

/// <summary>POST /api/extract: what the model read. Checks and flags are worked out in the browser (demoApi).</summary>
public sealed record LiveReading(ReceiptFields Fields, string? Category, double ExtractionSeconds);

public static class LiveDemoEndpoints
{
    public const string VisitorPolicy = "visitor";

    public const string Busy = "The author's PC is busy; try again in a minute.";
    public const string BudgetUsed = "The live demo has read all it will for now; try the sample receipts.";
    public const string TooSlow = "Reading took too long; try again.";
    public const string CouldntRead = "The model couldn't read this photo; try another.";

    public static string OverLimit(int perHour) =>
        $"You've reached the demo's limit of {perHour} receipts an hour; try again later.";

    // Visitors show up in the log as a short code, not their address. The salt is new every run.
    private static readonly byte[] Salt = RandomNumberGenerator.GetBytes(16);

    public static IEndpointRouteBuilder MapLiveDemoEndpoints(this IEndpointRouteBuilder app)
    {
        app.MapGet("/api/live", (IOptions<OllamaOptions> ollama, IOptions<LiveDemoOptions> live) =>
            new LiveStatus(ollama.Value.Model, live.Value.PerVisitorPerHour > 0 ? live.Value.PerVisitorPerHour : null));

        app.MapPost("/api/extract", Extract)
            .DisableAntiforgery() // no cookies to protect; CORS decides which site may read the answer
            .RequireRateLimiting(VisitorPolicy);

        return app;
    }

    private static async Task<IResult> Extract(
        IFormFile file,
        IReceiptExtractor extractor,
        ReadGate gate,
        IOptions<LiveDemoOptions> options,
        ILoggerFactory logs,
        HttpContext http,
        CancellationToken ct)
    {
        if (file.Length is 0 or > UploadRules.MaxBytes)
            return Invalid($"File must be between 1 byte and {UploadRules.MaxBytes / 1024 / 1024} MB.");
        if (!UploadRules.AllowedFiles.ContainsKey(Path.GetExtension(file.FileName)))
            return Invalid("Only JPG and PNG photos can be read.");
        if (!await UploadRules.LooksLikeImageAsync(file, ct))
            return Invalid("That file isn't a valid JPG or PNG image.");

        // In memory only: the photo is never written anywhere, and it's gone when this request ends.
        var image = new byte[file.Length];
        await using (var stream = file.OpenReadStream())
            await stream.ReadExactlyAsync(image, ct);

        var (turn, refused) = await gate.WaitTurnAsync(ct);
        if (refused is ReadGate.Refusal.Busy)
            return Results.Problem(Busy, statusCode: StatusCodes.Status503ServiceUnavailable);
        if (refused is ReadGate.Refusal.BudgetUsed)
            return Results.Problem(BudgetUsed, statusCode: StatusCodes.Status429TooManyRequests);

        using var _ = turn;
        var log = logs.CreateLogger("LiveDemo");
        var visitor = Visitor(http);
        using var timeout = CancellationTokenSource.CreateLinkedTokenSource(ct);
        timeout.CancelAfter(options.Value.ReadTimeout);
        var stopwatch = Stopwatch.StartNew();
        try
        {
            var result = await extractor.ExtractAsync(image, [], timeout.Token);
            var seconds = Math.Round(stopwatch.Elapsed.TotalSeconds, 2);
            log.LogInformation("Read a {Kb} KB photo for visitor {Visitor} in {Seconds}s; {Left} of {Budget} reads left",
                image.Length / 1024, visitor, seconds, gate.Left, options.Value.Budget);
            return Results.Ok(new LiveReading(result.Fields, result.Category, seconds));
        }
        catch (OperationCanceledException) when (!ct.IsCancellationRequested)
        {
            log.LogWarning("Gave up on a photo for visitor {Visitor} after {Seconds}s", visitor, Math.Round(stopwatch.Elapsed.TotalSeconds));
            return Results.Problem(TooSlow, statusCode: StatusCodes.Status504GatewayTimeout);
        }
        catch (ExtractionException ex)
        {
            // The real reason names local addresses and setup steps: it's for my log, not for visitors.
            log.LogWarning("Couldn't read a photo for visitor {Visitor}: {Error}", visitor, ex.Message);
            return Results.Problem(CouldntRead, statusCode: StatusCodes.Status502BadGateway);
        }
    }

    private static IResult Invalid(string message) =>
        Results.ValidationProblem(new Dictionary<string, string[]> { ["file"] = [message] });

    private static string Visitor(HttpContext http) =>
        http.Connection.RemoteIpAddress is { } ip
            ? Convert.ToHexString(HMACSHA256.HashData(Salt, ip.GetAddressBytes()))[..6]
            : "unknown";
}
```

```csharp
using System.Threading.RateLimiting;
using Microsoft.AspNetCore.HttpOverrides;
using Microsoft.Extensions.Options;
using ReceiptExtractor.Api.Extraction;

namespace ReceiptExtractor.Api.LiveDemo;

/// <summary>
/// The live demo (scripts/live-demo.ps1): a separate, minimal app that only reads photos for the demo site, through
/// Tailscale Funnel. It registers no database and no storage, so nothing a visitor sends can be kept.
/// See docs/superpowers/specs/2026-09-28-live-demo-design.md.
/// </summary>
public static class LiveDemoApp
{
    /// <summary>True when started with --LiveDemo:Enabled=true (or the LiveDemo__Enabled environment variable).</summary>
    public static bool IsRequested(string[] args) =>
        new ConfigurationBuilder().AddEnvironmentVariables().AddCommandLine(args).Build().GetValue<bool>("LiveDemo:Enabled");

    /// <param name="contentRoot">Where appsettings.json is (default: next to the app's code). Tests pass an empty folder.</param>
    /// <param name="configure">Tests: swap services (e.g. a fake extractor) and use a test server.</param>
    public static WebApplication Build(string[] args, string? contentRoot = null, Action<WebApplicationBuilder>? configure = null)
    {
        var builder = WebApplication.CreateBuilder(new WebApplicationOptions
        {
            Args = args,
            ContentRootPath = contentRoot ?? AppContext.BaseDirectory,
        });
        var options = builder.Configuration.GetSection("LiveDemo").Get<LiveDemoOptions>() ?? new();

        builder.Logging.AddSimpleConsole(o =>
        {
            o.SingleLine = true;
            o.TimestampFormat = "HH:mm:ss ";
        });
        builder.Services.AddProblemDetails();
        builder.Services.Configure<LiveDemoOptions>(builder.Configuration.GetSection("LiveDemo"));
        builder.Services.AddOllamaExtractor(builder.Configuration);
        builder.Services.AddSingleton<ReadGate>();

        // Funnel hands us requests from this PC: trust its X-Forwarded-For (from loopback only, the default).
        builder.Services.Configure<ForwardedHeadersOptions>(o => o.ForwardedHeaders = ForwardedHeaders.XForwardedFor);
        builder.Services.AddCors(o => o.AddDefaultPolicy(p => p.WithOrigins(options.AllowedOrigin).WithMethods("GET", "POST")));
        builder.Services.AddRateLimiter(o =>
        {
            o.RejectionStatusCode = StatusCodes.Status429TooManyRequests;
            o.AddPolicy(LiveDemoEndpoints.VisitorPolicy, http => options.PerVisitorPerHour > 0
                ? RateLimitPartition.GetFixedWindowLimiter(http.Connection.RemoteIpAddress?.ToString() ?? "unknown",
                    _ => new FixedWindowRateLimiterOptions { PermitLimit = options.PerVisitorPerHour, Window = TimeSpan.FromHours(1) })
                : RateLimitPartition.GetNoLimiter("everyone"));
            o.OnRejected = (context, _) => new ValueTask(
                Results.Problem(LiveDemoEndpoints.OverLimit(options.PerVisitorPerHour), statusCode: StatusCodes.Status429TooManyRequests)
                    .ExecuteAsync(context.HttpContext));
        });

        configure?.Invoke(builder);

        var app = builder.Build();
        app.UseForwardedHeaders();
        app.UseCors();
        app.UseRateLimiter();
        app.Use((http, next) =>
        {
            http.Response.Headers.CacheControl = "no-store";
            return next(http);
        });
        app.MapLiveDemoEndpoints();

        app.Lifetime.ApplicationStarted.Register(() =>
        {
            var model = app.Services.GetRequiredService<IOptions<OllamaOptions>>().Value.Model;
            Console.WriteLine();
            Console.WriteLine($"  Live demo: reading photos with {model}, at most {options.Budget}; stops at {DateTime.Now + options.StopAfter:HH:mm}.");
            Console.WriteLine();

            var stop = new Timer(_ => app.Lifetime.StopApplication(), null, options.StopAfter, Timeout.InfiniteTimeSpan);
            app.Lifetime.ApplicationStopping.Register(stop.Dispose);
        });

        return app;
    }
}
```

```ts
// The live demo (docs/superpowers/specs/2026-09-28-live-demo-design.md): while the author runs scripts/live-demo.ps1,
// visitors' own photos are read on the author's PC (the API's LiveDemo mode, through Tailscale Funnel at VITE_LIVE_URL).
// This module only talks to it; demoApi decides what happens to the receipts.
import type { Category, ReceiptUpdate } from '../api'

export interface LiveStatus {
  model: string
  perVisitorPerHour: number | null
}

export interface LiveAnswer {
  fields: ReceiptUpdate
  extractionSeconds: number
}

export interface LiveReader {
  /** The author's PC, if it's reading photos right now; null when it isn't. Remembered for 30 seconds. */
  status(): Promise<LiveStatus | null>
  /**
   * Waits for this photo's turn (one at a time, so a visitor's batch takes one place in the PC's line), calls
   * `onStart` and sends it. `onStart` returning false skips it (null). Errors carry a message for the visitor.
   */
  read(file: File, onStart: () => boolean): Promise<LiveAnswer | null>
}

export const WENT_OFFLINE = "The author's PC went offline; try again later."

const STATUS_TIMEOUT_MS = 3000
const STATUS_CACHE_MS = 30_000

interface Reading {
  fields: Omit<ReceiptUpdate, 'category'>
  category: Category | null
  extractionSeconds: number
}

export function createLiveReader(
  baseUrl: string | undefined,
  fetchFn: typeof fetch = (input, init) => fetch(input, init),
): LiveReader | undefined {
  if (!baseUrl) return undefined
  const base = baseUrl.replace(/\/+$/, '')
  let cached: { at: number; value: LiveStatus | null } | undefined
  let line: Promise<unknown> = Promise.resolve()

  const remember = (value: LiveStatus | null) => {
    cached = { at: Date.now(), value }
    return value
  }

  async function status(): Promise<LiveStatus | null> {
    if (cached && Date.now() - cached.at < STATUS_CACHE_MS) return cached.value
    try {
      const response = await fetchFn(`${base}/api/live`, { signal: AbortSignal.timeout(STATUS_TIMEOUT_MS) })
      return remember(response.ok ? ((await response.json()) as LiveStatus) : null)
    } catch {
      return remember(null)
    }
  }

  async function send(file: File): Promise<LiveAnswer> {
    const body = new FormData()
    body.append('file', file)
    let response: Response
    try {
      response = await fetchFn(`${base}/api/extract`, { method: 'POST', body })
    } catch {
      remember(null)
      throw new Error(WENT_OFFLINE)
    }
    if (!response.ok) throw new Error(await reason(response, () => remember(null)))
    const reading = (await response.json()) as Reading
    return { fields: { ...reading.fields, category: reading.category }, extractionSeconds: reading.extractionSeconds }
  }

  function read(file: File, onStart: () => boolean): Promise<LiveAnswer | null> {
    const turn = line.then(() => (onStart() ? send(file) : null))
    line = turn.catch(() => undefined) // a failed read doesn't hold up the next one
    return turn
  }

  return { status, read }
}

/** The service's own message (ProblemDetails `detail`, or the first validation error); otherwise it's gone. */
async function reason(response: Response, wentOffline: () => void): Promise<string> {
  try {
    const problem = (await response.json()) as { detail?: string; errors?: Record<string, string[]> }
    const message = problem.detail ?? Object.values(problem.errors ?? {})[0]?.[0]
    if (message) return message
  } catch {
    // not JSON: Funnel's own error page when the service has stopped
  }
  wentOffline()
  return WENT_OFFLINE
}
```

- [ ] **Step 1: Write the script**

`scripts/live-demo.ps1`:

```powershell
<#
  Live demo: for a few hours, visitors of the demo site (receipt-tracker-demo.pages.dev) can have their own receipt
  photos read by the model on this PC. Nothing is stored. See docs/superpowers/specs/2026-09-28-live-demo-design.md.

    powershell -ExecutionPolicy Bypass -File scripts\live-demo.ps1

  Stops by itself after 3 hours (LiveDemo:StopAfter), or press Ctrl+C. Either way, Funnel is turned off again.
  One-time setup (Tailscale, Funnel, VITE_LIVE_URL in Cloudflare Pages): README, "Read on the author's PC".
#>
param([int]$Port = 5090)

$ErrorActionPreference = 'Stop'
$root = Resolve-Path (Join-Path $PSScriptRoot '..')
$project = Join-Path $root 'src\ReceiptExtractor.Api'
$model = (Get-Content (Join-Path $project 'appsettings.json') -Raw | ConvertFrom-Json).Ollama.Model

# 1. Ollama is running and has the model.
try {
    $tags = Invoke-RestMethod 'http://localhost:11434/api/tags' -TimeoutSec 5
} catch {
    Write-Host "Can't reach Ollama on this PC. Start the Ollama app, then run this again."
    exit 1
}
if (-not ($tags.models.name -contains $model)) {
    Write-Host "The model $model isn't downloaded. Run scripts\pull-model.ps1, then run this again."
    exit 1
}

# 2. Tailscale is installed and signed in.
if (-not (Get-Command tailscale -ErrorAction SilentlyContinue)) {
    Write-Host "Tailscale isn't installed. Get it from https://tailscale.com/download, sign in, then run this again."
    exit 1
}
$tailscale = & tailscale status --json | ConvertFrom-Json
if ($tailscale.BackendState -ne 'Running') {
    Write-Host "Tailscale isn't signed in. Open Tailscale and sign in, then run this again."
    exit 1
}
$address = 'https://' + $tailscale.Self.DNSName.TrimEnd('.')

# 3. Build first, so the tunnel never opens onto nothing.
dotnet build $project -c Release --nologo -v q
if ($LASTEXITCODE -ne 0) { exit 1 }

# 4. Open the tunnel. The first time, this prints a link to allow Funnel on your tailnet.
& tailscale funnel --bg $Port
if ($LASTEXITCODE -ne 0) {
    Write-Host "Funnel didn't start. If it printed a link, open it to allow Funnel, then run this again."
    exit 1
}

Write-Host ""
Write-Host "  Live demo is on at $address"
Write-Host "  (VITE_LIVE_URL in Cloudflare Pages must be this address.) Press Ctrl+C to stop early."
Write-Host ""
try {
    dotnet run --project $project -c Release --no-build --no-launch-profile -- --urls "http://localhost:$Port" --LiveDemo:Enabled=true
} finally {
    & tailscale funnel --https=443 off | Out-Null
    Write-Host "Live demo stopped; Funnel is off."
}
```
